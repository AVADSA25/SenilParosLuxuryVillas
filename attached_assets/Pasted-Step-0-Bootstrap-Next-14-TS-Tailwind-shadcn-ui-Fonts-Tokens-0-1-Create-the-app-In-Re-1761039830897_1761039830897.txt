Step 0 — Bootstrap (Next 14 + TS + Tailwind + shadcn/ui + Fonts + Tokens)
0.1 Create the app
# In Replit shell
npx create-next-app@latest senil \
  --typescript --eslint --app --src-dir false --tailwind false \
  --import-alias "@/*"

cd senil

0.2 Install deps
# Tailwind toolchain
npm i -D tailwindcss postcss autoprefixer
npx tailwindcss init -p

# UI & utilities
npm i framer-motion lenis class-variance-authority tailwind-merge clsx \
      next-themes lucide-react

# shadcn/ui
npx shadcn@latest init -d
# (accept defaults; it will create components.json)

# Minimal components we’ll actually use
npx shadcn@latest add button card dialog input label form

import type { Config } from "tailwindcss"

  darkMode: ["class"],
  content: [
    "./app/**/*.{ts,tsx}",

    "./ui/**/*.{ts,tsx}",
  ],
  theme: {
    container: { center: true, padding: "1rem", screens: { "2xl": "1200px" } },

      colors: {

        olive: "var(--olive)",       // Olive Slate #75776A

  },

}
export default config

0.4 Global styles & brand tokens

app/globals.css

@tailwind base;
@tailwind components;
@tailwind utilities;

/* Brand tokens */
:root{
  --ash: #D8D5CC;        /* primary light */
  --olive: #75776A;      /* accents */
  --graphite: #1E1E1C;   /* text on light / dark bg */
  --pine: #4f6e47;       /* your stronger green (optional) */

  --border: #CFCBC2;     /* ash -15% */
  --card: #F2F1EE;       /* ash +6% white */
  --muted: #E6E4DD;      /* lighter ash for sections */
}

/* Base look */
html, body { height: 100% }
body { @apply bg-ash text-graphite antialiased; }

a { @apply underline-offset-4 decoration-olive hover:text-olive; }

/* Focus ring (keyboard) */
:focus-visible {
  outline: 2px solid var(--olive);
  outline-offset: 2px;
}

0.5 Fonts (Inter + Lora)

0.6 Base layout

app/layout.tsx

import "./globals.css"
import { inter, lora } from "./fonts"
import type { Metadata } from "next"

export const metadata: Metadata = {
  title: "SENIL — Luxurious Paros Villas | Price on Request",
  description: "Three villas, one horizon. Isterni, Paros — viewings by appointment.",
}

export default function RootLayout({ children }: { children: React.ReactNode }) {
  return (
    <html lang="en" suppressHydrationWarning>
      <body className={`${inter.variable} ${lora.variable} font-sans min-h-screen bg-ash text-graphite`}>
        <div id="root">{children}</div>
      </body>
    </html>
  )
}
